let langClient = null;

exports.activate = function() {
	let juliaPath = nova.workspace.config.get("julia.executablePath") 
				 || nova.config.get("julia.executablePath") 
				 || "/usr/local/bin/julia";

	// 1. Smart Project Path Detection
	let projectPath = nova.workspace.path;
	
	if (!projectPath && nova.workspace.activeTextEditor) {
		let docPath = nova.workspace.activeTextEditor.document.path;
		if (docPath) {
			projectPath = nova.path.dirname(docPath);
		}
	}
	
	// Fallback if absolutely no path is found
	projectPath = projectPath || "@.";

	// 2. Configure the Server (Isolated Environment)
	let serverOptions = {
		path: juliaPath,
		args: [
			"--startup-file=no",
			"--history-file=no",
			"--quiet",
			"-e",
			// Pass the target project path internally rather than via the executable flag
			`using LanguageServer; runserver(stdin, stdout, ENV["NOVA_JULIA_LSP_PATH"])`
		],
		type: "stdio",
		env: {
			// Safely pass the path to Julia without quote-escaping issues
			"NOVA_JULIA_LSP_PATH": projectPath
		}
	};

	let clientOptions = {
		syntaxes: ["julia"],
		debug: nova.inDevMode() 
	};

	langClient = new LanguageClient("julia-lsp", "Julia Language Server", serverOptions, clientOptions);

	try {
		langClient.start();
		nova.subscriptions.add(langClient);
	} catch (err) {
		console.error("Failed to start Julia Language Server:", err);
	}
}

class JuliaTaskAssistant {
	resolveTaskAction(context) {
		// Retrieve the current executable path
		let juliaPath = nova.workspace.config.get("julia.executablePath") 
					 || nova.config.get("julia.executablePath") 
					 || "/usr/local/bin/julia";

		let action = new TaskProcessAction(juliaPath);
		
		// Find the currently active document
		let activeFile = null;
		if (nova.workspace.activeTextEditor) {
			activeFile = nova.workspace.activeTextEditor.document.path;
		}

		if (!activeFile) {
			nova.workspace.showErrorMessage("No active Julia file to run.");
			return null;
		}

		// Run the script using the project environment of the workspace
		let projectPath = nova.workspace.path || "@.";
		
		action.args = [
			`--project=${projectPath}`,
			activeFile
		];
		
		return action;
	}
}

// Register the Run Task Assistant
nova.assistants.registerTaskAssistant(new JuliaTaskAssistant(), {
	identifier: "julia-tasks"
});


nova.commands.register("julia-lsp.restart", (editor) => {
	exports.deactivate();
	exports.activate();
});

exports.deactivate = function() {
	if (langClient) {
		langClient.stop();
		nova.subscriptions.remove(langClient);
		langClient = null;
	}
}
